cmake_minimum_required(VERSION 3.16.3)
project(agro_task_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BTCPP_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BTCPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BTCPP_GROOT_INTERFACE OFF CACHE BOOL "" FORCE)
set(BTCPP_SQLITE_LOGGING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BTCPP_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(behaviortree_cpp
 URL https://codeload.github.com/BehaviorTree/BehaviorTree.CPP/tar.gz/refs/tags/4.7.0
 URL_HASH SHA256=ee71a20daa245b4a8eb27c8352b0cb144831c456bdac4ed894694a1f78e407da)
FetchContent_MakeAvailable(behaviortree_cpp)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(agro_bt src/bridge.cpp src/executor.cpp)
target_include_directories(agro_bt PUBLIC include)
target_link_libraries(agro_bt PUBLIC BT::behaviortree_cpp CURL::libcurl Threads::Threads)
target_compile_options(agro_bt PRIVATE -Wall -Wextra -Wpedantic)
add_executable(agro-bt src/main.cpp)
target_link_libraries(agro-bt PRIVATE agro_bt)
install(TARGETS agro-bt RUNTIME DESTINATION bin)
enable_testing()
set(AGRO_CPP_TEST_SOURCE "" CACHE FILEPATH "External temporary verification source")
set(AGRO_CPP_TEST_REGISTRY "" CACHE FILEPATH "Registry for external verification")
if(AGRO_CPP_TEST_SOURCE)
 add_executable(agro_bt_verify ${AGRO_CPP_TEST_SOURCE})
 target_link_libraries(agro_bt_verify PRIVATE agro_bt)
 target_compile_options(agro_bt_verify PRIVATE -UNDEBUG)
 foreach(mode nonblocking halt unknown http500 mismatch missingerror badcancel)
  add_test(NAME agro_bt_${mode} COMMAND agro_bt_verify ${AGRO_CPP_TEST_REGISTRY} ${mode})
 endforeach()
endif()
